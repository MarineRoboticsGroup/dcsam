178 void updateContinuous(const gtsam::Values& continuousVals) {
179 for (const gtsam::Key& k : continuousKeys_) {
180 // If key `k` is not set continuousVals, skip it.
181 if (!continuousVals.exists(k)) continue;
182
183 if (continuousVals_.exists(k)) {
184 // If key `k` is set in stored continuousVals_, update its value
185 continuousVals_.update(k, continuousVals.at(k));
186 } else {
187 // If key `k` is not in the stored continuousVals_, create a new entry
188 // with key `k` and set its value to the one specified in the argument
189 // `continuousVals`
190 continuousVals_.insert(k, continuousVals.at(k));
191 }
192 }
193 }

195 void updateDiscrete(const DiscreteValues& discreteVals) {
196 for (const gtsam::DiscreteKey& dk : discreteKeys_) {
197 const gtsam::Key k = dk.first;
198 if (discreteVals.find(k) != discreteVals.end())
199 discreteVals_[k] = discreteVals.at(k);
200 }
201 }

209 }

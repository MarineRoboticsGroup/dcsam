159 const gtsam::Values& continuousVals,
160 [[maybe_unused]] const DiscreteValues& discreteVals) const {
161 gtsam::DecisionTreeFactor converted;
162 for (const gtsam::DiscreteKey& dkey : discreteKeys_) {
163 std::vector<double> probs = evalProbs(dkey, continuousVals);
164 // Cardinality of gtsam::DiscreteKey is located at `second`
165 assert(probs.size() == dkey.second);
166 gtsam::DecisionTreeFactor unary(dkey, probs);
167 converted = converted * unary;
168 }
169 return converted;
170 }

194 template <typename NonlinearFactorType>
195 double nonlinearFactorLogNormalizingConstant(
196 const NonlinearFactorType& factor, const gtsam::Values& values) const {
197 // Information matrix (inverse covariance matrix) for the factor.
198 gtsam::Matrix infoMat;
199
200 // NOTE: This is sloppy, is there a cleaner way?
201 std::shared_ptr<NonlinearFactorType> fPtr =
202 std::make_shared<NonlinearFactorType>(factor);
203 std::shared_ptr<NonlinearFactorType> factorPtr(fPtr);
204
205 // If this is a NoiseModelFactor, we'll use its noiseModel to
206 // otherwise noiseModelFactor will be nullptr
207 std::shared_ptr<gtsam::NoiseModelFactor> noiseModelFactor =
208 std::dynamic_pointer_cast<gtsam::NoiseModelFactor>(factorPtr);
209 if (noiseModelFactor) {
210 // If dynamic cast to NoiseModelFactor succeeded, see if the noise model
211 // is Gaussian
212 gtsam::noiseModel::Base::shared_ptr noiseModel =
213 noiseModelFactor->noiseModel();
214
215 std::shared_ptr<gtsam::noiseModel::Gaussian> gaussianNoiseModel =
216 std::dynamic_pointer_cast<gtsam::noiseModel::Gaussian>(noiseModel);
217 if (gaussianNoiseModel) {
218 // If the noise model is Gaussian, retrieve the information matrix
219 infoMat = gaussianNoiseModel->information();
220 } else {
221 // If the factor is not a Gaussian factor, we'll linearize it to get
222 // something with a normalized noise model
223 // TODO(kevin): does this make sense to do? I think maybe not in
224 // general? Should we just yell at the user?
225 std::shared_ptr<gtsam::GaussianFactor> gaussianFactor =
226 factor.linearize(values);
227 infoMat = gaussianFactor->information();
228 }
229 }
230
231 // Compute the (negative) log of the normalizing constant
232 return (factor.dim() * log(2.0 * M_PI) / 2.0) -
233 (log(infoMat.determinant()) / 2.0);
234 }

256 std::vector<double> evalProbs(const gtsam::DiscreteKey& dk,
257 const gtsam::Values& continuousVals) const {
258 std::vector<double> logProbs;
259 for (size_t i = 0; i < dk.second; i++) {
260 DiscreteValues testDiscreteVals;
261 testDiscreteVals[dk.first] = i;
262 // Recall: `error` returns -log(prob), so we compute exp(-error) to
263 // recover probability
264 double logProb = -error(continuousVals, testDiscreteVals);
265 logProbs.push_back(logProb);
266 }
267 return expNormalize(logProbs);
268 }
